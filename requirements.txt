# Convenience installer for local development.
# Installs both distributions in editable mode with every optional backend
# (redis, postgres, gui) and the test tools.
#
#   python -m pip install -r requirements.txt
#
# Equivalent to:
#   python -m pip install -e "./client[gui,dev]" -e "./server[all,dev]"

-e ./client[gui,dev]
-e ./server[all,dev]
